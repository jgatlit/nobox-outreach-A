Backend API Requirements (on Replit)
For this implementation to work, your Replit app should have the following API endpoints:

Authentication Endpoint (/auth)
javascript// In your Replit Express app
app.post('/auth', (req, res) => {
  // Validate API key/credentials
  const { apiKey } = req.body;
  if (apiKey !== process.env.API_KEY) {
    return res.status(401).json({ error: 'Invalid credentials' });
  }
  
  // Generate JWT token
  const token = jwt.sign(
    { authorized: true },
    process.env.JWT_SECRET,
    { expiresIn: '1h' }
  );
  
  res.json({ token });
});

Client Domains Endpoint (/domains)
javascript// In your Replit Express app
app.get('/domains', authenticateToken, async (req, res) => {
  try {
    // Connect to PostgreSQL
    const client = new Client({
      connectionString: process.env.DATABASE_URL,
    });
    await client.connect();
    
    // Execute PostgreSQL query
    const query = `
      SELECT lead_id AS "leadID", client_domain AS "clientDomain"
      FROM clients
      ORDER BY client_domain ASC
    `;
    const result = await client.query(query);
    await client.end();
    
    // Return results
    res.json(result.rows);
  } catch (error) {
    console.error('Database error:', error);
    res.status(500).json({ error: 'Database error' });
  }
});

Selection Endpoint (/select-domain)
javascript// In your Replit Express app
app.post('/select-domain', authenticateToken, async (req, res) => {
  try {
    const { leadID, timestamp } = req.body;
    
    // Connect to PostgreSQL
    const client = new Client({
      connectionString: process.env.DATABASE_URL,
    });
    await client.connect();
    
    // Execute PostgreSQL query to record the selection
    const query = `
      INSERT INTO domain_selections (lead_id, timestamp)
      VALUES ($1, $2)
      RETURNING id
    `;
    const result = await client.query(query, [leadID, timestamp]);
    await client.end();
    
    // Return the result
    res.json({ 
      success: true, 
      selectionId: result.rows[0].id 
    });
  } catch (error) {
    console.error('Database error:', error);
    res.status(500).json({ error: 'Database error' });
  }
});

Authentication Middleware
javascript// Middleware to verify JWT token
function authenticateToken(req, res, next) {
  const authHeader = req.headers['authorization'];
  const token = authHeader && authHeader.split(' ')[1];
  
  if (!token) {
    return res.status(401).json({ error: 'Authentication required' });
  }
  
  jwt.verify(token, process.env.JWT_SECRET, (err, user) => {
    if (err) {
      return res.status(403).json({ error: 'Invalid or expired token' });
    }
    
    req.user = user;
    next();
  });
}
